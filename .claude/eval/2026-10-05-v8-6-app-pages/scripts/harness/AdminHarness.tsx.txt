// Answer every /api call in the page before it leaves; anything unmocked gets a 503, never the network.
if (typeof window !== "undefined" && !(window as unknown as { __adminMocked?: boolean }).__adminMocked) {
  (window as unknown as { __adminMocked?: boolean }).__adminMocked = true;
  const orig = window.fetch.bind(window);
  window.fetch = (input: RequestInfo | URL, init?: RequestInit) => {
    const url = typeof input === "string" ? input : input instanceof URL ? input.href : input.url;
    if (!url.includes("/api/")) return orig(input, init);
    const body = adminMock(url, init);
    const status = body === undefined ? 503 : 200;
    return Promise.resolve(new Response(JSON.stringify(body ?? { error: "harness: no mock" }), { status, headers: { "content-type": "application/json" } }));
  };
}

const COURSE_PARAMS = Promise.resolve({ courseId: "course-earth" });
const PAGES: Record<string, () => React.ReactNode> = {
  "overview": () => <Overview />,
  "users": () => <Users />,
  "courses": () => <Courses />,
  "courses/course-earth": () => <CourseDetail params={COURSE_PARAMS} />,
  "courses/new": () => <CourseNew />,
  "knowledge-graph": () => <KnowledgeGraph />,
  "rag": () => <Rag />,
  "misconceptions": () => <Misconceptions />,
  "quiz-analytics": () => <QuizAnalytics />,
  "cost": () => <Cost />,
  "moderation": () => <Moderation />,
  "lesson-cache": () => <LessonCache />,
  "audit-log": () => <AuditLog />,
  "system": () => <System />,
};

export function AdminHarness({ slug }: { slug: string }) {
  const render = PAGES[slug] ?? PAGES.overview;
  return (
    <PathnameContext.Provider value={`/admin/${slug}`}>
      <AdminShell admin={{ email: "admin@example.com", full_name: "Robin Admin" }}>
        <Suspense fallback={null}>{render()}</Suspense>
      </AdminShell>
    </PathnameContext.Provider>
  );
}
