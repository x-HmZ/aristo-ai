const BLOOMS = ["remember", "understand", "apply", "analyze", "evaluate", "create"] as const;

/** ISO timestamp `offset` days after 2026-09-07 at the given UTC hour. */
function at(offset: number, hour = 10, minute = 0): string {
  return new Date(Date.UTC(2026, 8, 7 + offset, hour, minute)).toISOString();
}

/** YYYY-MM-DD `offset` days after 2026-09-07. */
function day(offset: number): string {
  return at(offset).slice(0, 10);
}

const round2 = (n: number) => Math.round(n * 100) / 100;
const round4 = (n: number) => Math.round(n * 10000) / 10000;

function concept(
  domain: string, id: string, name: string, bloom: string, difficulty: number, minutes: number,
  prerequisites: string[], terms: string[], misconception: string,
): MockConcept {
  return {
    id, name, domain,
    description:           `${name}: a middle-school lesson in ${domain.replace(/_/g, " ")}.`,
    difficulty,
    bloom_level:           bloom,
    estimated_minutes:     minutes,
    key_terms:             terms,
    learning_objectives:   [`Explain ${name.toLowerCase()} in your own words`, `Use ${name.toLowerCase()} in a short task`],
    common_misconceptions: [misconception],
    tags:                  [domain, bloom],
    prerequisites,
  };
}

function kgDomain(domain: string): Json {
  return { concepts: CONCEPTS_BY_DOMAIN[domain] ?? [] };
}

function kgCoverage(domain: string): Json {
  const concepts = CONCEPTS_BY_DOMAIN[domain] ?? [];
  const per: Record<string, { chunkCount: number; lastIngestedAt: string | null }> = {};
  concepts.forEach((c, i) => {
    const n = CHUNK_COUNTS[c.id] ?? 0;
    if (n > 0) per[c.id] = { chunkCount: n, lastIngestedAt: at(20 + (i % 6), 9 + i) };
  });
  const total = Object.values(per).reduce((s, c) => s + c.chunkCount, 0);
  return {
    domain_chunk_count: total + 3,
    per_concept:        per,
    last_ingested_at:   concepts.length ? at(26, 15) : null,
  };
}

function kgValidate(domain: string): Json {
  if (domain === PY) {
    return {
      ok:     true,
      valid:  false,
      errors: [
        { type: "orphan",               message: "Concept has no dependents and is not a leaf goal.", involved: ["py_dicts"] },
        { type: "missing_prerequisite", message: "Prerequisite id not found in this domain.",          involved: ["py_algorithms", "py_sorting"] },
      ],
    };
  }
  return { ok: true, valid: true, errors: [] };
}

function usersList(q: URLSearchParams): Json {
  const search = (q.get("search") ?? q.get("q") ?? "").trim().toLowerCase();
  const expertise = q.get("expertise");
  const status = q.get("status");
  const rows = USERS.filter((u) =>
    (!search || u.full_name.toLowerCase().includes(search) || u.email.includes(search)) &&
    (!expertise || u.expertise_level === expertise) &&
    (!status || u.approval_status === status)
  );
  return { users: rows, total: rows.length };
}

function structureFrom(concepts: MockConcept[], perModule: number): NonNullable<MockCourse["structure"]> {
  const modules: NonNullable<MockCourse["structure"]>["modules"] = [];
  for (let i = 0; i < concepts.length; i += perModule) {
    const slice = concepts.slice(i, i + perModule);
    const n = modules.length + 1;
    modules.push({
      id:          `m${n}`,
      title:       `Module ${n}: ${slice[0].name}`,
      description: `Covers ${slice.map((c) => c.name).join(", ")}.`,
      lessons:     slice.map((c) => ({ id: `l-${c.id}`, title: c.name, concept_ids: [c.id] })),
    });
  }
  return { modules };
}

function ragChunks(q: URLSearchParams): Json {
  const domainFilter = q.get("domain");
  const term = (q.get("q") ?? "").toLowerCase();
  const all = CHUNKS.map((c, i) => ({
    id:           `chunk-${String(i + 1).padStart(3, "0")}`,
    domain:       c.cid ? conceptDomain(c.cid) : i < 4 ? PY : EARTH,
    source_title: c.src,
    content:      c.text,
    metadata:     c.cid ? { concept_id: c.cid, chunk_index: i } : { chunk_index: i },
    created_at:   at(18 + i, 10),
  }));
  const chunks = all.filter((c) =>
    (!domainFilter || c.domain === domainFilter) && (!term || c.content.toLowerCase().includes(term))
  );
  return { chunks, total: chunks.length, domains: [EARTH, PY] };
}

function costSummary(range: string): Json {
  const scale = range === "24h" ? 0.15 : range === "30d" ? 3.6 : 1;
  const feats = FEATURE_COSTS.map((f) => ({ ...f, usd: round4(f.usd * scale), count: Math.max(1, Math.round(f.count * scale)) }));
  const total = round4(feats.reduce((s, f) => s + f.usd, 0));
  const events = feats.reduce((s, f) => s + f.count, 0);

  const sumBy = <K extends "provider" | "model" | "feature">(key: K) => {
    const m = new Map<string, { usd: number; count: number }>();
    for (const f of feats) {
      const cur = m.get(f[key]) ?? { usd: 0, count: 0 };
      m.set(f[key], { usd: round4(cur.usd + f.usd), count: cur.count + f.count });
    }
    return Array.from(m.entries()).sort((a, b) => b[1].usd - a[1].usd);
  };

  const days = range === "30d" ? 30 : 14;
  const wave = [1.0, 1.3, 0.8, 1.6, 1.1, 0.6, 0.9, 1.4, 1.8, 1.2, 0.7, 1.5, 1.9, 1.0];
  const trendBase = total / days;
  const trend = Array.from({ length: days }, (_, i) => ({
    day:   day(30 - days + i),
    usd:   round4(trendBase * wave[i % wave.length]),
    count: Math.round((events / days) * wave[i % wave.length]),
  }));

  const shares = [0.31, 0.22, 0.16, 0.12, 0.09, 0.06, 0.04];
  const breakdownUsers = [USERS[1], USERS[0], USERS[7], USERS[8], USERS[3], USERS[5], USERS[9]];
  const userBreakdown = breakdownUsers.map((u, i) => {
    const share = shares[i];
    const uf = feats.filter((_, k) => (k + i) % 3 !== 2).map((f) => ({ feature: f.feature, provider: f.provider, model: f.model, usd: round4(f.usd * share), count: Math.max(1, Math.round(f.count * share)) }));
    const group = (key: "feature" | "provider" | "model") => {
      const m = new Map<string, { usd: number; count: number }>();
      for (const f of uf) {
        const cur = m.get(f[key]) ?? { usd: 0, count: 0 };
        m.set(f[key], { usd: round4(cur.usd + f.usd), count: cur.count + f.count });
      }
      return Array.from(m.entries()).sort((a, b) => b[1].usd - a[1].usd);
    };
    return {
      user_id:    u.id,
      email:      i === 6 ? null : u.email,
      usd:        round4(uf.reduce((s, f) => s + f.usd, 0)),
      count:      uf.reduce((s, f) => s + f.count, 0),
      byFeature:  group("feature").map(([feature, v]) => ({ feature, ...v })),
      byProvider: group("provider").map(([provider, v]) => ({ provider, ...v })),
      byModel:    group("model").map(([model, v]) => ({ model, ...v })),
    };
  });

  return {
    range:                 range === "24h" || range === "30d" ? range : "7d",
    total_usd:             total,
    event_count:           events,
    projected_monthly_usd: round2(range === "24h" ? total * 30 : range === "30d" ? total : (total / 7) * 30),
    byProvider:            sumBy("provider").map(([provider, v]) => ({ provider, usd: v.usd })),
    byFeature:             sumBy("feature").map(([feature, v]) => ({ feature, ...v })),
    byModel:               sumBy("model").map(([model, v]) => ({ model, ...v })),
    trend,
    topUsers:              userBreakdown.slice(0, 5).map(({ user_id, email, usd, count }) => ({ user_id, email, usd, count })),
    userBreakdown,
    providerFeatureMatrix: sumBy("provider").map(([provider, v]) => ({
      provider,
      usd:      v.usd,
      features: feats.filter((f) => f.provider === provider).map(({ feature, usd, count }) => ({ feature, usd, count })),
    })),
  };
}

function nonGet(path: string): Json {
  let m: RegExpMatchArray | null;
  if ((m = path.match(/^\/api\/kg\/validate\/([^/]+)$/))) return kgValidate(decodeURIComponent(m[1]));
  if ((m = path.match(/^\/api\/kg\/generate\/([^/]+)$/))) {
    const concepts = CONCEPTS_BY_DOMAIN[decodeURIComponent(m[1])] ?? PYTHON_CONCEPTS;
    return { ok: true, count: concepts.length, concepts };
  }
  if (path === "/api/kg/concepts")        return { ok: true, inserted: 0, edges: 0 };
  if (path === "/api/admin/rag/test")     return { ok: true, matches: [] };
  if (path === "/api/admin/rag/upload")   return { ok: true, ingested: 0, total_chunks: 0 };
  return { ok: true };
}

export function adminMock(url: string, init?: RequestInit): unknown | undefined {
  let parsed: URL;
  try {
    parsed = new URL(url, "http://hx.local");
  } catch {
    return undefined;
  }
  const path = parsed.pathname.replace(/\/+$/, "");
  const q = parsed.searchParams;
  if (!path.startsWith("/api/admin/") && !path.startsWith("/api/kg/")) return undefined;

  const method = (init?.method ?? "GET").toUpperCase();
  if (method !== "GET" && method !== "HEAD") return nonGet(path);

  let m: RegExpMatchArray | null;
  switch (path) {
    case "/api/admin/analytics":         return overview();
    case "/api/admin/users":             return usersList(q);
    case "/api/admin/courses":           return { courses: COURSES };
    case "/api/admin/rag/chunks":        return ragChunks(q);
    case "/api/admin/misconceptions":    return misconceptions(q);
    case "/api/admin/quiz-analytics":    return quizAnalytics();
    case "/api/admin/cost/summary":      return costSummary(q.get("range") ?? "7d");
    case "/api/admin/moderation/queue":  return moderationQueue();
    case "/api/admin/lesson-cache":      return lessonCache(q);
    case "/api/admin/audit-log":         return auditLog(q);
    case "/api/admin/system/health":     return systemHealth();
    case "/api/kg/concepts":             return { concepts: ALL_CONCEPTS };
  }

  if ((m = path.match(/^\/api\/admin\/users\/([^/]+)$/)))               return userDetail(decodeURIComponent(m[1]));
  if ((m = path.match(/^\/api\/admin\/courses\/([^/]+)\/analytics$/))) return courseAnalytics(decodeURIComponent(m[1]));
  if ((m = path.match(/^\/api\/admin\/courses\/([^/]+)$/))) {
    const id = decodeURIComponent(m[1]);
    return { course: COURSES.find((c) => c.id === id) ?? COURSES[0] };
  }
  if ((m = path.match(/^\/api\/admin\/kg\/coverage\/([^/]+)$/)))        return kgCoverage(decodeURIComponent(m[1]));
  if ((m = path.match(/^\/api\/kg\/validate\/([^/]+)$/)))               return kgValidate(decodeURIComponent(m[1]));
  if ((m = path.match(/^\/api\/kg\/([^/]+)$/)))                         return kgDomain(decodeURIComponent(m[1]));

  return undefined;
}
