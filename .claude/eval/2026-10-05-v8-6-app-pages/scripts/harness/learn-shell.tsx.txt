// LOCAL ONLY (excluded from git): V8.4c harness. Mounts the real LearnClient (the /learn shell) with every
// /api call answered by an in-page mock, so the top bar, mode picker, free mode, course panel, desk quiz,
// daily review and lightbox can be captured headless in both themes. Nothing leaves the browser: any /api
// path without a mock rejects. The capture scripts also abort /api/** at the network layer.
//
//   ?v=loading | ?v=stalled         the loading screen alone
//   ?courses=empty                  ModePicker with no courses
//   ?gen=fail | ?gen=hang           what "Generate" answers with (never the real route)
//   ?review=empty | hang            what /api/quiz/review answers with (default: two questions)

function installMock(q: URLSearchParams) {
  const w = window as unknown as { __mocked?: boolean; __calls: string[]; __store: unknown };
  w.__store = useAristoStore;
  if (w.__mocked) return;
  w.__mocked = true;
  w.__calls = [];
  const orig = window.fetch.bind(window);
  const json = (b: unknown, status = 200) => Promise.resolve(new Response(JSON.stringify(b), { status, headers: { "content-type": "application/json" } }));
  const never = () => new Promise<Response>(() => {});
  const volcano = DEMO_TOPICS[0];
  window.fetch = (input: RequestInfo | URL, init?: RequestInit) => {
    const url = typeof input === "string" ? input : input instanceof URL ? input.href : input.url;
    if (!url.includes("/api/")) return orig(input, init);
    w.__calls.push(url);
    if (/\/api\/courses\/generate/.test(url)) return q.get("gen") === "hang" ? never() : json({ error: "mock" }, 500);
    if (/\/api\/courses\/c-/.test(url)) {
      // V8.6: ?map=fail (network error), ?map=empty (no modules), default: every mastery tier and a current concept.
      const m = q.get("map");
      if (m === "fail") return Promise.reject(new TypeError("harness: map failed"));
      if (m === "empty") return json({ course: { ...COURSES[0], structure: { modules: [] } }, masteryMap: {}, conceptMeta: {} });
      return json({ course: MAP_COURSE, masteryMap: MAP_MASTERY, conceptMeta: MAP_META });
    }
    if (/\/api\/auth\/onboarding/.test(url)) return json({ error: "harness" }, 500);
    if (/\/api\/profile\/analytics/.test(url)) {
      // V8.6: ?dash=empty (a new learner), ?dash=fail (network error), ?dash=hang (loading), default: everything filled.
      const d = q.get("dash");
      if (d === "fail") return Promise.reject(new TypeError("harness: analytics failed"));
      if (d === "hang") return never();
      if (d === "empty") return json({ streak: 0, time_today_seconds: 0, time_week_seconds: 0, total_concepts_seen: 0, total_mastered: 0, reviews_due: 0, weak_concepts: [], learner_profile: null });
      return json(DASH_FULL);
    }
    if (/\/api\/courses$/.test(url)) return json({ courses: q.get("courses") === "empty" ? [] : COURSES });
    if (/\/api\/profile$/.test(url)) return json({});
    if (/\/api\/profile\/session/.test(url)) return json({ ok: true });
    if (/overdue-count/.test(url)) return json({ overdueCount: 3 });
    if (/\/api\/learn\/lesson\//.test(url)) return json(volcano.lesson);
    if (/\/api\/quiz\/lesson\//.test(url)) return json({ questions: volcano.quiz });
    if (/\/api\/quiz\/review/.test(url)) {
      const r = q.get("review");
      if (r === "hang") return never();
      if (r === "empty") return json({ questions: [], conceptCount: 0 });
      return json({ questions: volcano.quiz, conceptCount: 2 });
    }
    if (/\/api\/quiz\/submit/.test(url)) {
      const b = JSON.parse(String(init?.body ?? "{}"));
      const ok = String(b.userAnswer).trim().toLowerCase() === String(b.correctAnswer).trim().toLowerCase();
      return new Promise((res) => setTimeout(() => res(new Response(JSON.stringify({ is_correct: ok, score: ok ? 1 : 0, feedback: ok ? "Right." : "The correct answer is " + b.correctAnswer + ".", misconception_detected: null }), { headers: { "content-type": "application/json" } })), 400));
    }
    if (/\/api\/quiz\/complete/.test(url)) return json({ ok: true });
    if (/\/api\/learn\/complete/.test(url)) return json({ ok: true });
    // No generated visuals in the harness: the lesson plays without images.
    if (/\/api\/learn\/segment-visuals/.test(url)) return json({ results: [] });
    // Narration: a pre-rendered demo clip from /public (static, no ElevenLabs call).
    if (/\/api\/tts/.test(url)) return orig("/demo/volcano-eruption/seg_001.mp3");
    // Anything else (narration from /api/tts included) gets an error response, never the real route.
    return json({ error: "harness: blocked" }, 503);
  };
}

export default function LearnShellHarness() {
  const [view, setView] = useState<string | null>(null);
  useEffect(() => {
    const q = new URLSearchParams(location.search);
    installMock(q);
    setView(q.get("v") ?? "shell");
  }, []);
  if (process.env.NODE_ENV === "production") return <div>404</div>;
  if (!view) return null;
  if (view === "loading" || view === "stalled") {
    return (
      <div className="fixed inset-0 z-[100]">
        <LoadingScreenVisual progress={40} stalled={view === "stalled"} />
      </div>
    );
  }
  return <LearnClient userName="Sam" userId="harness-user" onboardingDone={new URLSearchParams(location.search).get("onboarded") !== "0"} domain={null} />;
}
